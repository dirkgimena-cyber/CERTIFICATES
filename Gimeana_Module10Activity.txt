# KIOSK_PROTOTYPE
# CAFETERIA DATA AND MENU
menu = ["Spicy Chicken Sandwich w/ Fries", "Cheeseburger", "Garden Salad"]
prices = [150, 120, 90]

# THE MENU
print("===== CAFETERIA KIOSK =====")
print("0.", menu[0], "- PHP", prices[0])
print("1.", menu[1], "- PHP", prices[1])
print("2.", menu[2], "- PHP", prices[2])

# USER INPUT
try:
    choice = int(input("Type the NUMBER of your food choice: "))

    # SELECTED FOOD AND PRICE
    if choice >= 0 and choice < len(menu):
        print("\nYou selected:", menu[choice])
        print("Price: PHP", prices[choice])
    else:
        print("Invalid choice. Please choose 0, 1, or 2.")

except ValueError:
    print("Invalid input. Please enter a number.")
